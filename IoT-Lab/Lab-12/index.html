3. Example ESP8266 Arduino Code

This optional reference demonstrates a simple Wi-Fi web server controlling two low-voltage LEDs as appliance indicators. Replace the indicators with suitable isolated hardware only under instructor supervision.

On an ESP8266 development board, GPIO5 is often labelled D1 and GPIO4 is often labelled D2. Confirm your board’s pinout.

#include <ESP8266WiFi.h>

const char* ssid = "YOUR_WIFI_NAME";
const char* password = "YOUR_WIFI_PASSWORD";
WiFiServer server(80);
const int lightPin = 5; // GPIO5 (often D1)
const int fanPin = 4;   // GPIO4 (often D2)

void setup() {
  Serial.begin(115200);
  pinMode(lightPin, OUTPUT);
  pinMode(fanPin, OUTPUT);
  digitalWrite(lightPin, LOW);
  digitalWrite(fanPin, LOW);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  server.begin();
  Serial.println(WiFi.localIP());
}

void loop() {
  WiFiClient client = server.available();
  if (!client) return;
  client.setTimeout(1000);
  String request = client.readStringUntil('\r');
  if (request.indexOf("GET /light/on ") >= 0) digitalWrite(lightPin, HIGH);
  if (request.indexOf("GET /light/off ") >= 0) digitalWrite(lightPin, LOW);
  if (request.indexOf("GET /fan/on ") >= 0) digitalWrite(fanPin, HIGH);
  if (request.indexOf("GET /fan/off ") >= 0) digitalWrite(fanPin, LOW);
  client.println("HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nConnection: close\r\n\r\nCommand received");
  client.stop();
}

Example commands for a real, correctly configured ESP8266: http://DEVICE_IP/light/on and http://DEVICE_IP/fan/off. This simple teaching example has no authentication; do not expose it to the public internet.
